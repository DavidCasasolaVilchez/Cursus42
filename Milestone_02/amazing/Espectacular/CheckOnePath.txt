    def has_multiple_paths(self) -> bool:
        """Return True when there are at least two paths from start to end."""
        directions: list[tuple[Cardinal, int, int, bool]] = [
            (Cardinal.N, -1, 0, False),
            (Cardinal.E, 0, 1, False),
            (Cardinal.S, 1, 0, False),
            (Cardinal.W, 0, -1, False),
        ]

        walls = {
            Cardinal.N: "north",
            Cardinal.E: "east",
            Cardinal.S: "south",
            Cardinal.W: "west",
        }

        start = self.start_cell.position
        end = self.end_cell.position

        paths_found = 0
        stack: list[tuple[tuple[int, int], set[tuple[int, int]]]] = [
            (start, {start})
        ]

        while stack:
            position, visited = stack.pop()

            if position == end:
                paths_found += 1
                if paths_found > 1:
                    return True
                continue

            y, x = position
            current_cell = self.map[y][x]

            for direction, delta_y, delta_x, _ in directions:
                if getattr(current_cell.state, walls[direction]):
                    continue

                next_y = y + delta_y
                next_x = x + delta_x

                if not (
                    0 <= next_y < self.size[0]
                    and 0 <= next_x < self.size[1]
                ):
                    continue

                next_cell = self.map[next_y][next_x]
                next_position = (next_y, next_x)

                if next_cell.type == TypeCell.CLOSED42:
                    continue

                if next_position in visited:
                    continue

                stack.append((
                    next_position,
                    visited | {next_position}
                ))

        return False










 @staticmethod
    def has_deadend_or_3x3_hole(panel: "Panel") -> bool:
        directions: list[tuple[int, int, str]] = [
            (-1, 0, "north"),
            (0, 1, "east"),
            (1, 0, "south"),
            (0, -1, "west"),
        ]

        for y in range(panel.size[0]):
            for x in range(panel.size[1]):
                cell = panel.map[y][x]

                if cell.type == TypeCell.CLOSED42:
                    continue

                open_paths = 0

                for offset_y, offset_x, wall in directions:
                    next_y = y + offset_y
                    next_x = x + offset_x

                    if not (
                        0 <= next_y < panel.size[0]
                        and 0 <= next_x < panel.size[1]
                    ):
                        continue

                    next_cell = panel.map[next_y][next_x]

                    if next_cell.type == TypeCell.CLOSED42:
                        continue

                    if not getattr(cell.state, wall):
                        open_paths += 1

                if open_paths == 1:
                    print(
                        f"Dead-end encontrado en la celda "
                        f"{cell.position}"
                    )
                    return True

        for y in range(panel.size[0] - 2):
            for x in range(panel.size[1] - 2):
                has_hole = True

                for block_y in range(y, y + 3):
                    for block_x in range(x, x + 3):
                        cell = panel.map[block_y][block_x]

                        if cell.type == TypeCell.CLOSED42:
                            has_hole = False
                            break

                    if not has_hole:
                        break

                if not has_hole:
                    continue

                for block_y in range(y, y + 3):
                    for block_x in range(x, x + 3):
                        cell = panel.map[block_y][block_x]

                        if block_x < x + 2 and cell.state.east:
                            has_hole = False
                            break

                        if block_y < y + 2 and cell.state.south:
                            has_hole = False
                            break

                    if not has_hole:
                        break

                if has_hole:
                    print(
                        f"Hueco de 3x3 encontrado desde la celda "
                        f"({y}, {x})"
                    )
                    return True

        return False
